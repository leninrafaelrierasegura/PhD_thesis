# grid in y
y <- seq(0, 1, length.out = 100)
n_time <- 50  # number of frames


df <- do.call(rbind, lapply(1:n_time, function(t) {
  data.frame(
    x = 0,
    y = y,
    z = sin(6*pi*y) * sin(2*pi * t / n_time),
    time = t
  )
}))

range_z <- range(df$z)

  frame = ~time,
  type = "scatter3d",
  mode = "lines",
  line = list(width = 6),
  color = ~z,
  colors = "RdBu"
)
